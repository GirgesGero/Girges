Nmap Network Analyzer

A Python-based network scanning and analysis tool that automates Nmap scans, parses results, and generates detailed reports for security assessments.

Technologies Used
PythonNmapNetworkingSecurity
Features
  • Automated network discovery and port scanning
  • Service version detection
  • Vulnerability reporting
  • Export results to multiple formats (JSON, CSV, HTML)
Code Sample
import nmap
import json
from datetime import datetime

class NetworkAnalyzer:
    def __init__(self):
        self.scanner = nmap.PortScanner()

    def scan_network(self, target, ports='1-1000'):
        # Perform comprehensive network scan
        self.scanner.scan(target, ports, arguments='-sV -sC -O')
        return self.parse_results()

    def parse_results(self):
        results = []
        for host in self.scanner.all_hosts():
            host_info = {
                'ip': host,
                'hostname': self.scanner[host].hostname(),
                'state': self.scanner[host].state(),
                'ports': []
            }
            for proto in self.scanner[host].all_protocols():
                for port in self.scanner[host][proto].keys():
                    service = self.scanner[host][proto][port]
                    host_info['ports'].append({
                        'port': port,
                        'protocol': proto,
                        'state': service['state'],
                        'service': service['name'],
                        'version': service.get('version', '')
                    })
            results.append(host_info)
        return results

    def export_report(self, results, fmt='json'):
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        filename = "scan_report_" + timestamp + "." + fmt
        with open(filename, 'w') as f:
            if fmt == 'json':
                json.dump(results, f, indent=2)
        print("Report saved:", filename)

# Usage example
analyzer = NetworkAnalyzer()
results = analyzer.scan_network('192.168.1.0/24')
analyzer.export_report(results, 'json')
